create or replace TABLE SDP_PROJECT_SANDBOX.DATATHON.CH06_USER_AFFINITYS (
	USER_ID VARCHAR(200),
	CONTENT_PILLAR VARCHAR(16777216),
	ARTICLE_COUNT NUMBER(18,0),
	AFFINITY_SCORE FLOAT,
	LAST_READ_AT DATE
);


create or replace TABLE SDP_PROJECT_SANDBOX.DATATHON.CH06_USER_DATA (
	USER_ID VARCHAR(50) NOT NULL COMMENT 'Primary key. Unique identifier for each user',
	ARTICLE_ID VARCHAR(50) COMMENT 'Unique identifier for each article',
	EMAIL VARCHAR(50) COMMENT 'User''s email address for sending The Round Up',
	COUNTRY VARCHAR(50) COMMENT 'User''s country, for regional content filtering',
	OPT_IN BOOLEAN COMMENT 'Whether the user has consented to receive emails',
	ENGAGED_USER BOOLEAN COMMENT 'Users having 2+ page views per week for 4 consecutive weeks',
	LAST_VISIT DATE COMMENT 'Date of the user''s most recent visit',
	M_AVG_TIME_SPENT_PER_USER TIME(9) COMMENT 'Time spent / user',
	CONTENT_PILLAR VARCHAR(50) COMMENT 'Content Pillar Name',
	CONTENT_TOPIC VARCHAR(150) COMMENT 'Topics included/related to the article',
	primary key (USER_ID)
);

create or replace TABLE SDP_PROJECT_SANDBOX.DATATHON.CH06_USER_ARTICLE_MAPPING (
	USER_ID VARCHAR(200) NOT NULL COMMENT 'Primary key. Unique identifier for each User',
	CONTENT_URL VARCHAR(16777216) NOT NULL COMMENT 'Primary key. Unique identifier for each Article',
	READ_AT TIMESTAMP_NTZ(9) COMMENT 'Timestamp that the customer opened the article',
	VISIT_DURATION NUMBER(38,0),
	primary key (USER_ID, CONTENT_URL)
);

create or replace TABLE SDP_PROJECT_SANDBOX.DATATHON.CH06_TOP_ARTICLES_BY_PILLAR (
	CONTENT_PILLAR VARCHAR(16777216),
	PILLAR_RANK NUMBER(18,0),
	"Article ID" VARCHAR(16777216),
	DATE_PUBLISHED TIMESTAMP_NTZ(9),
	PAGE VARCHAR(16777216),
	PAGE_TITLE VARCHAR(16777216),
	CONTENT_URL VARCHAR(16777216),
	PAGE_VIEWS NUMBER(18,0),
	UNIQUE_VISITORS NUMBER(18,0)
);

create or replace TABLE SDP_PROJECT_SANDBOX.DATATHON.CH06_ARTICLE_DATA (
	"Article ID" VARCHAR(16777216),
	CONTENT_PILLAR VARCHAR(16777216),
	CONTENT_URL VARCHAR(16777216),
	PAGE_TITLE VARCHAR(16777216),
	PAGE_VIEWS NUMBER(18,0)
);
